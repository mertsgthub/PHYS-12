#include 
#define BUTTON_PIN  D3
#define SERVO_LED_PIN  D4 

#define POT_PIN  A1     
#define LED_PIN  D2     
#define WIFI_CHANNEL 1

uint8_t receiverMac[] = { 0x58, 0x8C, 0x81, 0xAB, 0x36, 0x6C };

typedef struct {
  int potValue;     
  bool servoExtended;   
} Packet;

Packet outgoing;
esp_now_peer_info_t peerInfo;

int           lastReading  = HIGH;
int           buttonState  = HIGH;
unsigned long lastDebounce = 0;
const unsigned long debounceMs = 50;

void onDataSent(const wifi_tx_info_t *info, esp_now_send_status_t status) {
  digitalWrite(LED_PIN, status == ESP_NOW_SEND_SUCCESS ? HIGH : LOW);
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP); 
  pinMode(SERVO_LED_PIN, OUTPUT);
  digitalWrite(SERVO_LED_PIN, HIGH);         
  analogReadResolution(12);         

  WiFi.mode(WIFI_STA);        
  WiFi.setSleep(false);                                
  esp_wifi_set_channel(WIFI_CHANNEL, WIFI_SECOND_CHAN_NONE);          

  if (esp_now_init() != ESP_OK) { Serial.println("ESP-NOW init failed"); return; }
  esp_now_register_send_cb(onDataSent);

  memcpy(peerInfo.peer_addr, receiverMac, 6);
  peerInfo.channel = WIFI_CHANNEL;                
  peerInfo.encrypt = false;
  if (esp_now_add_peer(&peerInfo) != ESP_OK) { Serial.println("Failed to add peer"); return; }
}

void loop() {
  int reading = digitalRead(BUTTON_PIN);
  if (reading != lastReading) lastDebounce = millis();
  if (millis() - lastDebounce > debounceMs) {
    if (reading != buttonState) {
      buttonState = reading;
      if (buttonState == LOW) {               
        outgoing.servoExtended = !outgoing.servoExtended;
        digitalWrite(SERVO_LED_PIN, outgoing.servoExtended ? LOW : HIGH);   
      }
    }
  }
  lastReading = reading;
  outgoing.potValue = analogRead(POT_PIN);
  esp_now_send(receiverMac, (uint8_t *)&outgoing, sizeof(outgoing));
  delay(50);                           
}
