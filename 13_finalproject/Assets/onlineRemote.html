import serial, struct, threading, time
from flask import Flask, Response, request, render_template_string

PORT = "/dev/cu.usbmodem101"   # <-- change to your sender's port
BAUD = 921600
MAGIC = b'\xA5\x5A\xA5\x5A'

app = Flask(__name__)
ser = serial.Serial(PORT, BAUD, timeout=0.1)
serial_lock = threading.Lock()

latest_frame = None
frame_lock = threading.Lock()

def reader_thread():
    global latest_frame
    buf = bytearray()
    while True:
        with serial_lock:
            data = ser.read(4096)
        if data:
            buf += data
        while True:
            i = buf.find(MAGIC)
            if i < 0:
                if len(buf) > 200000: del buf[:-4]
                break
            if len(buf) < i + 8: break
            (n,) = struct.unpack('<I', bytes(buf[i+4:i+8]))
            if n == 0 or n > 200000:
                del buf[:i+4]; continue
            if len(buf) < i + 8 + n: break
            jpg = bytes(buf[i+8:i+8+n])
            del buf[:i+8+n]
            with frame_lock:
                latest_frame = jpg
        time.sleep(0.01)

threading.Thread(target=reader_thread, daemon=True).start()

def gen_mjpeg():
    while True:
        with frame_lock:
            frame = latest_frame
        if frame is not None:
            yield (b'--frame\r\nContent-Type: image/jpeg\r\n\r\n' + frame + b'\r\n')
        time.sleep(0.05)

@app.route('/stream')
def stream():
    return Response(gen_mjpeg(), mimetype='multipart/x-mixed-replace; boundary=frame')

@app.route('/speed', methods=['POST'])
def speed():
    pct = int(request.json.get('value', 0))
    pot = max(0, min(4095, int(pct * 4095 / 100)))
    with serial_lock:
        ser.write(f"P{pot}\n".encode())
    return {'ok': True}

@app.route('/servo', methods=['POST'])
def servo():
    state = int(request.json.get('state', 0))
    with serial_lock:
        ser.write(f"V{state}\n".encode())
    return {'ok': True}

  btn.addEventListener('click', () => {
    servoOn = !servoOn;
    btn.textContent = servoOn ? 'Retract (0°)' : 'Extend (180°)';
    btn.className = servoOn ? '' : 'off';
    fetch('/servo', {method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({state: servoOn ? 1 : 0})});

@app.route('/')
def index():
    return render_template_string(PAGE)

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5001, threaded=True)
