#include "esp_camera.h"
#include <WiFi.h>
#include <esp_now.h>
#include <esp_wifi.h>

#define WIFI_CHANNEL   1
#define CHUNK_SIZE     240     
#define FRAME_GAP_MS   100
#define CAM_FRAMESIZE  FRAMESIZE_QQVGA
#define CAM_QUALITY    15

uint8_t senderMac[] = { 0x94, 0xA9, 0x90, 0x6E, 0x2B, 0x60 };

#define PWDN_GPIO_NUM  32
#define RESET_GPIO_NUM -1
#define XCLK_GPIO_NUM   0
#define SIOD_GPIO_NUM  26
#define SIOC_GPIO_NUM  27
#define Y9_GPIO_NUM    35
#define Y8_GPIO_NUM    34
#define Y7_GPIO_NUM    39
#define Y6_GPIO_NUM    36
#define Y5_GPIO_NUM    21
#define Y4_GPIO_NUM    19
#define Y3_GPIO_NUM    18
#define Y2_GPIO_NUM     5
#define VSYNC_GPIO_NUM 25
#define HREF_GPIO_NUM  23
#define PCLK_GPIO_NUM  22

typedef struct __attribute__((packed)) {
  uint8_t  magic;
  uint16_t frameId;
  uint16_t chunkIdx;
  uint16_t chunkTotal;
  uint16_t len;
  uint8_t  data[CHUNK_SIZE];
} ImgChunk;

ImgChunk chunk;
uint16_t frameId = 0;
volatile bool txDone = true;
volatile uint32_t okCount = 0, failCount = 0;
unsigned long lastStat = 0;

void onSent(const wifi_tx_info_t *info, esp_now_send_status_t status) {
  if (status == ESP_NOW_SEND_SUCCESS) okCount++; else failCount++;
  txDone = true;
}

bool sendChunk(size_t payloadLen) {
  uint32_t t0 = millis();
  while (!txDone && millis() - t0 < 20) delay(1);
  txDone = false;
  if (esp_now_send(senderMac, (uint8_t *)&chunk, 9 + payloadLen) != ESP_OK) {
    txDone = true;
    return false;
  }
  return true;
}

void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println();
  Serial.println("===== CAM BOOT =====");

  camera_config_t c;
  c.ledc_channel = LEDC_CHANNEL_0;
  c.ledc_timer   = LEDC_TIMER_0;
  c.pin_d0 = Y2_GPIO_NUM;   c.pin_d1 = Y3_GPIO_NUM;
  c.pin_d2 = Y4_GPIO_NUM;   c.pin_d3 = Y5_GPIO_NUM;
  c.pin_d4 = Y6_GPIO_NUM;   c.pin_d5 = Y7_GPIO_NUM;
  c.pin_d6 = Y8_GPIO_NUM;   c.pin_d7 = Y9_GPIO_NUM;
  c.pin_xclk = XCLK_GPIO_NUM;   c.pin_pclk  = PCLK_GPIO_NUM;
  c.pin_vsync = VSYNC_GPIO_NUM; c.pin_href  = HREF_GPIO_NUM;
  c.pin_sccb_sda = SIOD_GPIO_NUM; c.pin_sccb_scl = SIOC_GPIO_NUM;
  c.pin_pwdn = PWDN_GPIO_NUM;   c.pin_reset = RESET_GPIO_NUM;
  c.xclk_freq_hz = 20000000;
  c.pixel_format = PIXFORMAT_JPEG;
  c.frame_size   = CAM_FRAMESIZE;
  c.jpeg_quality = CAM_QUALITY;
  c.fb_count     = psramFound() ? 2 : 1;
  c.fb_location  = psramFound() ? CAMERA_FB_IN_PSRAM : CAMERA_FB_IN_DRAM;
  c.grab_mode    = CAMERA_GRAB_LATEST;

  if (esp_camera_init(&c) != ESP_OK) {
    Serial.println("Camera init FAILED");
    while (true) delay(1000);
  }
  Serial.println("Camera OK");

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  delay(200);                                  // let the Wi-Fi stack come up
  WiFi.setSleep(false);
  esp_wifi_set_channel(WIFI_CHANNEL, WIFI_SECOND_CHAN_NONE);

  uint8_t mac[6];
  esp_wifi_get_mac(WIFI_IF_STA, mac);
  Serial.printf("CAM MAC: %02X:%02X:%02X:%02X:%02X:%02X\n",
                mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  Serial.printf("Sending to: %02X:%02X:%02X:%02X:%02X:%02X\n",
                senderMac[0], senderMac[1], senderMac[2],
                senderMac[3], senderMac[4], senderMac[5]);

  bool allZero = true;
  for (int i = 0; i < 6; i++) if (senderMac[i] != 0) allZero = false;
  if (allZero) Serial.println(">>> ERROR: senderMac[] is still all zeros! <<<");

  if (esp_now_init() != ESP_OK) { Serial.println("ESP-NOW init FAILED"); return; }
  esp_now_register_send_cb(onSent);

  esp_now_peer_info_t peer = {};
  memcpy(peer.peer_addr, senderMac, 6);
  peer.channel = WIFI_CHANNEL;
  peer.encrypt = false;
  if (esp_now_add_peer(&peer) != ESP_OK) Serial.println("add_peer FAILED");
  else Serial.println("Peer added OK");
  Serial.println("====================");
}

void loop() {
  camera_fb_t *fb = esp_camera_fb_get();
  if (!fb) { delay(50); return; }

  uint16_t total = (fb->len + CHUNK_SIZE - 1) / CHUNK_SIZE;
  for (uint16_t i = 0; i < total; i++) {
    size_t off = (size_t)i * CHUNK_SIZE;
    size_t n   = min((size_t)CHUNK_SIZE, fb->len - off);
    chunk.magic      = 0xA5;
    chunk.frameId    = frameId;
    chunk.chunkIdx   = i;
    chunk.chunkTotal = total;
    chunk.len        = n;
    memcpy(chunk.data, fb->buf + off, n);
    sendChunk(n);
  }

  esp_camera_fb_return(fb);
  frameId++;

  if (millis() - lastStat > 2000) {
    lastStat = millis();
    Serial.printf("frame %u | %u bytes | sent OK:%lu FAIL:%lu\n",
                  frameId, (unsigned)fb->len, (unsigned long)okCount, (unsigned long)failCount);
    okCount = failCount = 0;
  }

  delay(FRAME_GAP_MS);
}
